4. async testProcess() {
	let res= await this.getPrStMode()
	switch res.mode:
		case 0: this.stop({ua:`Зупинено користувачем`, en:...}); return False
		case 2: this.error({ua:`Аварія!`, en:...}); return False
		case 4,5,6,7: this.error({ua:`Йде автоналаштування!`, en:...}); return False
	if (res.Pr.value == this.goal.Pr && res.St.value == this.goal.St) return False
	else this.finish({ua:"Крок виконано успішно", en:..})
	setTimeout(() => this.testProcess(), this.checkPeriod * 1000);
}

async beforeAll(){
